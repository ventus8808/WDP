# 1) Create and activate a conda env with R + cmdstanr dependencies
conda create -y -n brms r-base=4.3 r-essentials
conda activate brms

# 2) Install R packages from CRAN/PPM inside R
R --vanilla <<'RS'
options(repos = c(CRAN = "https://packagemanager.posit.co/cran/__linux__/jammy/latest"))
install.packages(c(
  "data.table", "dplyr", "stringr", "tidyr", "readr", "purrr",
  "cmdstanr", "posterior", "optparse"
))
# cmdstanr recommends setting the C++ toolchain; on clusters with modules this might already be set.
# Validate toolchain and then install CmdStan (choose a version; here default latest).
library(cmdstanr)
# Set a persistent install path in your home to avoid re-installation per-job
install_dir <- file.path(path.expand("~"), ".cmdstan")
Sys.setenv(CMDSTANR_WRITE_INSTALL_DIR=install_dir)
if (!dir.exists(install_dir)) dir.create(install_dir, recursive = TRUE)
install_cmdstan(cores = parallel::detectCores())
# Show version
cat("CmdStan path:", cmdstan_path(), "\n")
RS
